class Solution {
    public boolean isPalindromic(String s) {

        String binary = "";
        for (int i = 0; i < s.length(); i++) {
           int ascii = s.charAt(i);
            String b = Integer.toBinaryString(ascii);
            while (b.length() < 8) {
                b = "0" + b;
            }
             binary = binary + b;
        }

        int left = 0;
        int right = binary.length() - 1;

        while (left < right) {
           if (binary.charAt(left) != binary.charAt(right)) {
                return false;
            }

            left++;
            right--;
        }
        return true;
    }
}

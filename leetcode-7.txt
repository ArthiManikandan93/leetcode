class Solution {
    public double minPrice(int[] prices, int[] discounts) {
        Arrays.sort(prices);
        Arrays.sort(discounts);
        double sum=0;
        int n=prices.length;
        int m=discounts.length;
        int c=Math.min(n,m);
        for(int i=0;i<c;i++){
            int price=prices[n-1-i];
            int dis=discounts[m-1-i];
            sum+=price*(100.0-dis)/100.0;
        }
        for(int i=c;i<n;i++){
            sum=sum+prices[n-1-i];
        }
        return sum;
    }
}

class Solution {
    public void sortColors(int[] nums) {
            
            int zerocount=0,count1=0,count2=0;
             for(int i=0;i<nums.length;i++){
                if(nums[i]==0)
                    zerocount++;
                    else if(nums[i]==1)
                    count1++;
                    else
                    count2++;

        }
        
        for(int i=0;i<zerocount;i++)
              nums[i]=0;
               
        for(int i=zerocount;i<zerocount+count1;i++)
              nums[i]=1;
          
        for(int i=zerocount+count1;i<nums.length;i++)
              nums[i]=2;
    }
}
